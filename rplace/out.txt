
pub struct User {
    pub name: String,
    pub id: i32,
}
impl User {
    pub fn get_name(&self) -> &String {
        &self.name
    } 
    pub fn get_id(&self) -> &i32 {
        &self.id
    } 

    pub fn set_name(&self, name: String){
        self.name = name;
    } 
    pub fn set_id(&self, id: i32){
        self.id = id;
    } 

}
